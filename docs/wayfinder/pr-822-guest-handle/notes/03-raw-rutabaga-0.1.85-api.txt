== Rutabaga::resource_create_blob (src/rutabaga_core.rs:950) ==
pub fn resource_create_blob(
    &mut self,
    ctx_id: u32,
    resource_id: u32,
    resource_create_blob: ResourceCreateBlob,
    iovecs: Option<Vec<RutabagaIovec>>,
    handle: Option<RutabagaHandle>,
) -> RutabagaResult<()> {
    if self.resources.contains_key(&resource_id) { return Err(RutabagaError::InvalidResourceId); }
    let component = self.components.get_mut(&self.default_component).ok_or(RutabagaError::InvalidComponent)?;
    let mut context = None;
    // For the cross-domain context ... Use `ctx_id` and the component type
    if ctx_id > 0 {
        let ctx = self.contexts.get_mut(&ctx_id).ok_or(RutabagaError::InvalidContextId)?;
        if ctx.component_type() == RutabagaComponentType::CrossDomain { context = Some(ctx); }
    }
    let resource = match context {
        Some(ctx) => ctx.context_create_blob(resource_id, resource_create_blob, handle)?,
        None => component.create_blob(ctx_id, resource_id, resource_create_blob, iovecs, handle)?,
    };
    self.resources.insert(resource_id, resource);
    Ok(())
}
